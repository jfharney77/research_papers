% =============================================================================
% NeurIPS Conference Paper Template
%
% IMPORTANT: This template requires the NeurIPS style file, which must be
% downloaded from the official NeurIPS website and placed in this directory:
%
%   https://neurips.cc/Conferences/2025/PaperInformation/StyleFiles
%
% Download neurips_2025.sty (or the current year's file), place it in:
%   latex/neurips/neurips_2025.sty
%
% Then update the \usepackage line below to match the filename.
%
% Compilation:
%   pdflatex main.tex
%   bibtex main
%   pdflatex main.tex
%   pdflatex main.tex
%
% Structure:
%   main.tex              -- root document (this file)
%   neurips_YYYY.sty      -- NeurIPS style file (download separately)
%   sections/abstract.tex
%   sections/introduction.tex
%   sections/related_work.tex
%   sections/conclusion.tex
%   references.bib        -- bibliography entries
% =============================================================================

\documentclass{article}

% ---------------- NeurIPS style ----------------
% Options:
%   (none)         submission mode -- anonymised, line numbers
%   [final]        accepted paper -- author names shown
%   [preprint]     non-anonymous preprint (e.g. arXiv)
%   [nonatbib]     disable natbib (if you prefer a different bib package)
\usepackage[final]{neurips_2025}   % <-- update year to match downloaded .sty

% ---------------- Packages ----------------
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{hyperref}
\usepackage{url}
\usepackage{booktabs}
\usepackage{amsfonts}
\usepackage{amsmath,amssymb}
\usepackage{nicefrac}
\usepackage{microtype}
\usepackage{xcolor}
\usepackage{graphicx}
% ------------------------------------------

\title{Paper Title Goes Here}

\author{%
  First Author \\
  Department of Computer Science \\
  University Name \\
  City, Country \\
  \texttt{first.author@university.edu}
  \And
  Second Author \\
  Department of Electrical Engineering \\
  University Name \\
  City, Country \\
  \texttt{second.author@university.edu}
}

\begin{document}

\maketitle

% ---------------- Abstract ----------------
\input{sections/abstract}

% ---------------- Body sections ----------------
\input{sections/introduction}
\input{sections/related_work}
\input{sections/figures}
\input{sections/conclusion}

% ---------------- References ----------------
\bibliographystyle{abbrvnat}
\bibliography{references}

\end{document}
